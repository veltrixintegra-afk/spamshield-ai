   273→   273→
   274→   274→function ThreatGauge({ score }: { score: number }) {
   275→   275→  const [animatedScore, setAnimatedScore] = useState(0)
   276→   276→  const color = getGaugeColor(score)
   277→   277→  const verdict = getVerdictInfo(score)
   278→   278→  const radius = 70
   279→   279→  const circumference = 2 * Math.PI * radius
   280→   280→  const offset = circumference - (animatedScore / 100) * circumference
   281→   281→
   282→   282→  useEffect(() => {
   283→   283→    let start = 0
   284→   284→    const duration = 1500
   285→   285→    const startTime = Date.now()
   286→   286→    const animate = () => {
   287→   287→      const elapsed = Date.now() - startTime
   288→   288→      const progress = Math.min(elapsed / duration, 1)
   289→   289→      const eased = 1 - Math.pow(1 - progress, 3)
   290→   290→      start = Math.round(eased * score)
   291→   291→      setAnimatedScore(start)
   292→   292→      if (progress < 1) requestAnimationFrame(animate)
   293→   293→    }
   294→   294→    requestAnimationFrame(animate)
   295→   295→  }, [score])
   296→   296→
   297→   297→  return (
   298→   298→    <div className="relative flex flex-col items-center gap-3">
   299→   299→      <div className="relative">
   300→   300→        <svg width="180" height="180" viewBox="0 0 180 180" className="transform -rotate-90">
   301→   301→          <circle
   302→   302→            cx="90"
   303→   303→            cy="90"
   304→   304→            r={radius}
   305→   305→            stroke="currentColor"
   306→   306→            strokeWidth="10"
   307→   307→            fill="none"
   308→   308→            className="text-muted/30"
   309→   309→          />
   310→   310→          <circle
   311→   311→            cx="90"
   312→   312→            cy="90"
   313→   313→            r={radius}
   314→   314→            stroke={color}
   315→   315→            strokeWidth="10"
   316→   316→            fill="none"
   317→   317→            strokeLinecap="round"
   318→   318→            strokeDasharray={circumference}
   319→   319→            strokeDashoffset={offset}
   320→   320→            style={{ transition: 'stroke-dashoffset 0.3s ease' }}
   321→   321→          />
   322→   322→        </svg>
   323→   323→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   324→   324→          <span className="text-4xl font-bold" style={{ color }}>
   325→   325→            {animatedScore}
   326→   326→          </span>
   327→   327→          <span className="text-xs text-muted-foreground uppercase tracking-wider">
   328→   328→            / 100
   329→   329→          </span>
   330→   330→        </div>
   331→   331→      </div>
   332→   332→      <Badge
   333→   333→        variant="outline"
   334→   334→        className={`${verdict.bgColor} ${verdict.borderColor} ${verdict.color} border px-4 py-1.5 text-sm font-semibold`}
   335→   335→      >
   336→   336→        {verdict.icon}
   337→   337→        <span className="ml-1.5">{verdict.label}</span>
   338→   338→      </Badge>
   339→   339→    </div>
   340→   340→  )
   341→   341→}
