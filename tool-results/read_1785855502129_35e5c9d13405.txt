   218→/* ============================================================
   219→   COMPONENT: THREAT GAUGE (SVG Circle)
   220→   ============================================================ */
   221→
   222→function ThreatGauge({ score }: { score: number }) {
   223→  const [animatedScore, setAnimatedScore] = useState(0)
   224→  const color = getGaugeColor(score)
   225→  const verdict = getVerdictInfo(score)
   226→  const radius = 70
   227→  const circumference = 2 * Math.PI * radius
   228→  const offset = circumference - (animatedScore / 100) * circumference
   229→
   230→  useEffect(() => {
   231→    let start = 0
   232→    const duration = 1500
   233→    const startTime = Date.now()
   234→    const animate = () => {
   235→      const elapsed = Date.now() - startTime
   236→      const progress = Math.min(elapsed / duration, 1)
   237→      const eased = 1 - Math.pow(1 - progress, 3)
   238→      start = Math.round(eased * score)
   239→      setAnimatedScore(start)
   240→      if (progress < 1) requestAnimationFrame(animate)
   241→    }
   242→    requestAnimationFrame(animate)
   243→  }, [score])
   244→
   245→  return (
   246→    <div className="relative flex flex-col items-center gap-3">
   247→      <div className="relative">
   248→        <svg width="180" height="180" viewBox="0 0 180 180" className="transform -rotate-90">
   249→          <circle
   250→            cx="90"
   251→            cy="90"
   252→            r={radius}
   253→            stroke="currentColor"
   254→            strokeWidth="10"
   255→            fill="none"
   256→            className="text-muted/30"
   257→          />
   258→          <circle
   259→            cx="90"
   260→            cy="90"
   261→            r={radius}
   262→            stroke={color}
   263→            strokeWidth="10"
   264→            fill="none"
   265→            strokeLinecap="round"
   266→            strokeDasharray={circumference}
   267→            strokeDashoffset={offset}
   268→            style={{ transition: 'stroke-dashoffset 0.3s ease' }}
   269→          />
   270→        </svg>
   271→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   272→          <span className="text-4xl font-bold" style={{ color }}>
   273→            {animatedScore}
   274→          </span>
   275→          <span className="text-xs text-muted-foreground uppercase tracking-wider">
   276→            / 100
   277→          </span>
   278→        </div>
   279→      </div>
   280→      <Badge
   281→        variant="outline"
   282→        className={`${verdict.bgColor} ${verdict.borderColor} ${verdict.color} border px-4 py-1.5 text-sm font-semibold`}
   283→      >
   284→        {verdict.icon}
   285→        <span className="ml-1.5">{verdict.label}</span>
   286→      </Badge>
   287→    </div>
   288→  )
   289→}
