   273→
   274→function ThreatGauge({ score }: { score: number }) {
   275→  const [animatedScore, setAnimatedScore] = useState(0)
   276→  const color = getGaugeColor(score)
   277→  const verdict = getVerdictInfo(score)
   278→  const radius = 70
   279→  const circumference = 2 * Math.PI * radius
   280→  const offset = circumference - (animatedScore / 100) * circumference
   281→
   282→  useEffect(() => {
   283→    let start = 0
   284→    const duration = 1500
   285→    const startTime = Date.now()
   286→    const animate = () => {
   287→      const elapsed = Date.now() - startTime
   288→      const progress = Math.min(elapsed / duration, 1)
   289→      const eased = 1 - Math.pow(1 - progress, 3)
   290→      start = Math.round(eased * score)
   291→      setAnimatedScore(start)
   292→      if (progress < 1) requestAnimationFrame(animate)
   293→    }
   294→    requestAnimationFrame(animate)
   295→  }, [score])
   296→
   297→  return (
   298→    <div className="relative flex flex-col items-center gap-3">
   299→      <div className="relative">
   300→        <svg width="180" height="180" viewBox="0 0 180 180" className="transform -rotate-90">
   301→          <circle
   302→            cx="90"
   303→            cy="90"
   304→            r={radius}
   305→            stroke="currentColor"
   306→            strokeWidth="10"
   307→            fill="none"
   308→            className="text-muted/30"
   309→          />
   310→          <circle
   311→            cx="90"
   312→            cy="90"
   313→            r={radius}
   314→            stroke={color}
   315→            strokeWidth="10"
   316→            fill="none"
   317→            strokeLinecap="round"
   318→            strokeDasharray={circumference}
   319→            strokeDashoffset={offset}
   320→            style={{ transition: 'stroke-dashoffset 0.3s ease' }}
   321→          />
   322→        </svg>
   323→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   324→          <span className="text-4xl font-bold" style={{ color }}>
   325→            {animatedScore}
   326→          </span>
   327→          <span className="text-xs text-muted-foreground uppercase tracking-wider">
   328→            / 100
   329→          </span>
   330→        </div>
   331→      </div>
   332→      <Badge
   333→        variant="outline"
   334→        className={`${verdict.bgColor} ${verdict.borderColor} ${verdict.color} border px-4 py-1.5 text-sm font-semibold`}
   335→      >
   336→        {verdict.icon}
   337→        <span className="ml-1.5">{verdict.label}</span>
   338→      </Badge>
   339→    </div>
   340→  )
   341→}

   397→
   398→  // ── Refs ──

   412→  // ── API: Load History ──
   413→  const loadHistory = async () => {
   414→    setIsLoadingHistory(true)
   415→    try {
   416→      const res = await fetch('/api/history')
   417→      if (!res.ok) throw new Error('Error al cargar historial')
   418→      const data = await res.json()
   419→      setHistory(data)
   420→    } catch {
   421→      toast.error('No se pudo cargar el historial de análisis')
   422→    } finally {
   423→      setIsLoadingHistory(false)
   424→    }
   425→  }
   426→
   427→  // ── API: Load Dashboard ──
   428→  const loadDashboard = async () => {
   429→    setIsLoadingDashboard(true)
   430→    try {
   431→      const res = await fetch('/api/dashboard')
   432→      if (!res.ok) throw new Error()
   433→      const data = await res.json()
   434→      setDashboard(data)
   435→      // Fetch AI insights separately (non-blocking)
   436→      if (data.totalScans >= 2 && !data.aiInsights) {
   437→        fetchAiInsights(data)
   438→      }
   439→    } catch {
   440→      // Silent fail
   441→    } finally {
   442→      setIsLoadingDashboard(false)
   443→    }
   444→  }
   445→
   446→  const fetchAiInsights = async (currentData: DashboardData) => {
   447→    try {
   448→      const res = await fetch('/api/ai-insights')
   449→      if (!res.ok) return
   450→      const { insights } = await res.json()
   451→      setDashboard({ ...currentData, aiInsights: insights })
   452→    } catch {
   453→      // Non-critical
   454→    }
   455→  }
   456→
   457→  // ── API: Scan Text ──
   458→  const handleScanText = async () => {
   459→    if (!emailContent.trim()) {
   460→      toast.error('Por favor, ingresa el contenido del correo electrónico')
   461→      return
   462→    }
   463→
   464→    setIsScanning(true)
   465→    setShowResults(true)
   466→    setScanResult(null)
   467→
   468→    try {
   469→      const res = await fetch('/api/scan', {
   470→        method: 'POST',
   471→        headers: { 'Content-Type': 'application/json' },
   472→        body: JSON.stringify({ emailContent }),
   473→      })
   474→
   475→      if (!res.ok) throw new Error('Error en el análisis')
   476→
   477→      const data: ScanResult = await res.json()

   479→      toast.success('Análisis completado exitosamente')

   482→      toast.error('Error al analizar el correo. Intenta de nuevo.')
   483→      setShowResults(false)
   484→    } finally {
   485→      setIsScanning(false)
   486→    }

   516→      toast.error('Error al analizar el archivo. Intenta de nuevo.')

   519→      setIsScanning(false)

   526→      toast.error('No hay resultados para generar el informe')
   527→      return
   528→    }
   529→
   530→    toast.info('Generando informe profesional...')
   531→
   532→    try {
   533→      const res = await fetch('/api/generate-report', {
   534→        method: 'POST',
   535→        headers: { 'Content-Type': 'application/json' },
   536→        body: JSON.stringify({ scanId: scanResult.id }),
   537→      })
   538→
   539→      if (!res.ok) throw new Error('Error al generar el informe')
   540→
   541→      const htmlContent = await res.text()
   542→      const blob = new Blob([htmlContent], { type: 'text/html' })
   543→      const url = URL.createObjectURL(blob)
   544→      const printWindow = window.open(url, '_blank')

   555→        a.click()

   557→        URL.revokeObjectURL(url)
   558→        toast.success('Informe descargado. Ábrelo y usa "Imprimir > Guardar como PDF".')

   580→      return
   581→    }
