   255→/* ============================================================
   256→   COMPONENT: THREAT GAUGE (SVG Circle)
   257→   ============================================================ */
   258→
   259→function ThreatGauge({ score }: { score: number }) {
   260→  const [animatedScore, setAnimatedScore] = useState(0)
   261→  const color = getGaugeColor(score)
   262→  const verdict = getVerdictInfo(score)
   263→  const radius = 70
   264→  const circumference = 2 * Math.PI * radius
   265→  const offset = circumference - (animatedScore / 100) * circumference
   266→
   267→  useEffect(() => {
   268→    let start = 0
   269→    const duration = 1500
   270→    const startTime = Date.now()
   271→    const animate = () => {
   272→      const elapsed = Date.now() - startTime
   273→      const progress = Math.min(elapsed / duration, 1)
   274→      const eased = 1 - Math.pow(1 - progress, 3)
   275→      start = Math.round(eased * score)
   276→      setAnimatedScore(start)
   277→      if (progress < 1) requestAnimationFrame(animate)
   278→    }
   279→    requestAnimationFrame(animate)
   280→  }, [score])
   281→
   282→  return (
   283→    <div className="relative flex flex-col items-center gap-3">
   284→      <div className="relative">
   285→        <svg width="180" height="180" viewBox="0 0 180 180" className="transform -rotate-90">
   286→          <circle
   287→            cx="90"
   288→            cy="90"
   289→            r={radius}
   290→            stroke="currentColor"
   291→            strokeWidth="10"
   292→            fill="none"
   293→            className="text-muted/30"
   294→          />
   295→          <circle
   296→            cx="90"
   297→            cy="90"
   298→            r={radius}
   299→            stroke={color}
   300→            strokeWidth="10"
   301→            fill="none"
   302→            strokeLinecap="round"
   303→            strokeDasharray={circumference}
   304→            strokeDashoffset={offset}
   305→            style={{ transition: 'stroke-dashoffset 0.3s ease' }}
   306→          />
   307→        </svg>
   308→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   309→          <span className="text-4xl font-bold" style={{ color }}>
   310→            {animatedScore}
   311→          </span>
   312→          <span className="text-xs text-muted-foreground uppercase tracking-wider">
   313→            / 100
   314→          </span>
   315→        </div>
   316→      </div>
   317→      <Badge
   318→        variant="outline"
   319→        className={`${verdict.bgColor} ${verdict.borderColor} ${verdict.color} border px-4 py-1.5 text-sm font-semibold`}
   320→      >
   321→        {verdict.icon}
   322→        <span className="ml-1.5">{verdict.label}</span>
   323→      </Badge>
   324→    </div>
   325→  )
   326→}
